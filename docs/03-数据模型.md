# 数据模型

## Rust 结构定义

```rust
pub struct Project {
    pub workdir: PathBuf,
    pub config: ProjectConfig,
    pub groups: Vec<Group>,
}

/// 一个 Group = config/ 下一个子目录 = 一个 Excel 文件
pub struct Group {
    pub name: String,
    pub dir: PathBuf,
    pub tables: Vec<Table>,
    pub constants: Vec<Constant>,
}

/// Table = 行列数据配置
pub struct Table {
    pub name: String,
    pub path: PathBuf,
    pub schema: TableSchema,
    pub record_count: usize,
}

/// Table 的表头定义
pub struct TableSchema {
    pub fields: Vec<FieldDef>,  // 第一列固定为 id (主键)
}

pub struct FieldDef {
    pub name: String,
    pub desc: String,
    pub tbl_type: String,
    pub export: Export,
}

/// Constant = 离散常量配置
pub struct Constant {
    pub name: String,
    pub path: PathBuf,
    pub entries: Vec<ConstEntry>,
}

pub struct ConstEntry {
    pub name: String,
    pub tbl_type: String,
    pub value: String,
    pub export: Export,
    pub desc: String,
}

pub enum Export {
    ClientServer,  // 前后端
    ClientOnly,    // 客户端
    ServerOnly,    // 服务器
    None,          // 不导出
}
```
